cmake_minimum_required(VERSION 3.24)
project(padcode_controller LANGUAGES C CXX)
include(FetchContent)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SDL_AUDIO OFF CACHE BOOL "" FORCE)
set(SDL_VIDEO OFF CACHE BOOL "" FORCE)
set(SDL_RENDER OFF CACHE BOOL "" FORCE)
set(SDL_GPU OFF CACHE BOOL "" FORCE)
set(SDL_CAMERA OFF CACHE BOOL "" FORCE)
set(SDL_HAPTIC OFF CACHE BOOL "" FORCE)
set(SDL_SENSOR OFF CACHE BOOL "" FORCE)
# This helper never opens a window; Linux must not require X11/Wayland headers.
set(SDL_UNIX_CONSOLE_BUILD ON CACHE BOOL "" FORCE)
FetchContent_Declare(SDL3
  URL https://github.com/libsdl-org/SDL/releases/download/release-3.4.16/SDL3-3.4.16.tar.gz
  URL_HASH SHA256=7322236cd12090c3eb40b9728be4d49c76f66ad17d04369584d4ecad5cf77c68
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(SDL3)
add_executable(padcode-controller main.cpp)
target_compile_features(padcode-controller PRIVATE cxx_std_17)
target_compile_options(padcode-controller PRIVATE -Wall -Wextra -Werror)
target_link_libraries(padcode-controller PRIVATE SDL3::SDL3-static)
if(APPLE)
  enable_language(OBJC)
  add_executable(padcode-bluetooth bluetooth.m permission.m discovery.m)
  target_compile_options(padcode-bluetooth PRIVATE -fobjc-arc -Wall -Wextra -Werror)
  target_link_libraries(padcode-bluetooth PRIVATE "-framework Foundation" "-framework IOBluetooth" "-framework CoreBluetooth")
  target_link_options(padcode-bluetooth PRIVATE "LINKER:-sectcreate,__TEXT,__info_plist,${CMAKE_CURRENT_SOURCE_DIR}/bluetooth.plist")
endif()

if(APPLE)
  set_property(TARGET padcode-bluetooth APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/bluetooth.plist")
endif()
